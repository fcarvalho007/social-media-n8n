export async function getConfigRevista(edicaoId: string): Promise<ConfigRevista> {
  const { data, error } = await supabase
    .from("revista_edicao").select("*").eq("edicao_id", edicaoId).maybeSingle();
  if (error) throw error;
  if (data) return data as ConfigRevista;
  // Os blocos fixos (livro e serviços) nascem preenchidos: ligados sem
  // conteúdo não apareceriam na pré-visualização.
  const { data: novo, error: e2 } = await supabase
    .from("revista_edicao")
    .insert({ edicao_id: edicaoId, ...VALORES_NOVA_EDICAO_REVISTA })
    .select("*").single();

  if (e2) throw e2;
  return novo as ConfigRevista;
}

export async function guardarConfigRevista(
  edicaoId: string, patch: Partial<Omit<ConfigRevista, "edicao_id">>,
): Promise<void> {
  const { error } = await supabase.from("revista_edicao").update(patch).eq("edicao_id", edicaoId);
  if (error) throw error;
}

export async function getItensRevista(edicaoId: string): Promise<ItemRevista[]> {
  const { data, error } = await supabase
    .from("revista_itens")
    .select("*, noticia:noticias(id, titulo, descricao, url, url_curto, categoria)")
    .eq("edicao_id", edicaoId)
    .order("ordem", { ascending: true });
  if (error) throw error;
  return (data ?? []).map((r) => ({
    ...r,
    noticia: Array.isArray(r.noticia) ? (r.noticia[0] ?? null) : r.noticia,
  })) as ItemRevista[];
}

/**
 * Universo único de notícias da edição — mesma definição usada pelo composer,
 * pelo cálculo de «Só site», pelos contadores e pela versão web.
 */
export async function getAprovadasDaEdicao(edicaoId: string): Promise<NoticiaAprovada[]> {
  const linhas = await carregarNoticiasDaEdicao(supabase as never, edicaoId);
  return linhas as unknown as NoticiaAprovada[];
}



export async function adicionarItem(
  edicaoId: string, noticiaId: string, papel: PapelRevista,
): Promise<void> {
  const itens = await getItensRevista(edicaoId);
  if (itens.some((i) => i.noticia_id === noticiaId)) {
    throw new Error("Essa notícia já está a ser usada nesta edição.");
  }
  const doPapel = itens.filter((i) => i.papel === papel);
  if (papel === "destaque" && doPapel.length >= MAXIMOS[papel]) {
    throw new Error(papel === "destaque"
      ? `«${ROTULOS_REVISTA.destaques}» já tem ${LIMITES_REVISTA.destaquesMax} notícias.`
      : "");
  }
  const noticia = (await supabase
    .from("noticias").select("descricao").eq("id", noticiaId).maybeSingle()).data;
  const { error } = await supabase.from("revista_itens").insert({
    edicao_id: edicaoId,
    noticia_id: noticiaId,
    papel,
    ordem: doPapel.length,
    resumo_factual: papel === "destaque" ? ((noticia?.descricao ?? "") as string) : "",
  });
  if (error) throw error;
}

export async function removerItem(itemId: string): Promise<void> {
  const { error } = await supabase.from("revista_itens").delete().eq("id", itemId);
  if (error) throw error;
}

export async function actualizarItem(
  itemId: string,
  patch: Partial<Pick<ItemRevista, "titulo_override" | "resumo_factual" | "minha_leitura" | "cta_rotulo" | "radar_nota">>,
): Promise<void> {
  const { error } = await supabase.from("revista_itens").update(patch).eq("id", itemId);
  if (error) throw error;
}

/** Move um item entre secções, respeitando os limites e sem duplicar. */
export async function moverItemPapel(
  edicaoId: string, itemId: string, papel: PapelRevista,
): Promise<void> {
  const itens = await getItensRevista(edicaoId);
  const doPapel = itens.filter((i) => i.papel === papel && i.id !== itemId);
  if (papel === "destaque" && doPapel.length >= MAXIMOS[papel]) {
    throw new Error(papel === "destaque"
      ? `«${ROTULOS_REVISTA.destaques}» já tem ${LIMITES_REVISTA.destaquesMax} notícias.`
      : "");
  }
  const { error } = await supabase
    .from("revista_itens").update({ papel, ordem: doPapel.length }).eq("id", itemId);
  if (error) throw error;
}

export async function reordenarItens(ids: string[]): Promise<void> {
  await Promise.all(ids.map((id, i) =>
    supabase.from("revista_itens").update({ ordem: i }).eq("id", id)));
}
