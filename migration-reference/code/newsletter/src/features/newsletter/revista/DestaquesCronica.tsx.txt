// «Destaques da crónica»: frase em grande e número da semana, escolhidos do texto ou escritos à mão.

import { useState } from "react";
import { toast } from "sonner";
import { Quote, Hash, TextSelect, Trash2 } from "lucide-react";
import { paragrafosCronica as dividirParagrafos, posicaoValida } from "@/lib/newsletter-engine/revista/sequencia-cronica";
import type { ConfigRevista } from "./data-revista";
import { pecaDeTrecho, ROTULO_ORIGEM, type OrigemPeca, type TipoTrecho } from "./pecas-trecho";

export type OrigensDestaque = Partial<Record<TipoTrecho | "lede", OrigemPeca | null>>;

const CAMPOS: Record<TipoTrecho, (keyof ConfigRevista)[]> = {
  pull_quote: ["pull_quote", "pull_quote_posicao"],
  momento: ["momento_activo", "momento_etiqueta", "momento_valor", "momento_descricao", "momento_posicao"],
};

const input = "w-full rounded-lg border border-input bg-background px-3 py-2 text-[14px] text-foreground outline-none focus:border-primary disabled:opacity-50";

export function DestaquesCronica({
  cfg, editar, excerto, bloqueado, origens, definirOrigem,
}: {
  cfg: ConfigRevista; editar: (p: Partial<ConfigRevista>) => void; excerto: string; bloqueado: boolean;
  origens: OrigensDestaque; definirOrigem: (o: OrigensDestaque) => void;
}) {
  const pars = dividirParagrafos(excerto);
  const [escolher, setEscolher] = useState<TipoTrecho | null>(null);

  const guardar = (tipo: TipoTrecho) =>
    Object.fromEntries(CAMPOS[tipo].map((k) => [k, cfg[k]])) as Partial<ConfigRevista>;

  const comDesfazer = (tipo: TipoTrecho, patch: Partial<ConfigRevista>, origem: OrigemPeca | null, msg: string) => {
    const antes = guardar(tipo); const origemAntes = origens[tipo] ?? null;
    editar(patch); definirOrigem({ [tipo]: origem });
    toast.success(msg, { duration: 8000, action: { label: "Desfazer", onClick: () => { editar(antes); definirOrigem({ [tipo]: origemAntes }); } } });
  };

  const aMao = (tipo: TipoTrecho, p: Partial<ConfigRevista>) => {
    editar(p); if (origens[tipo] !== "mao") definirOrigem({ [tipo]: "mao" });
  };

  const usarSelecao = (tipo: TipoTrecho, paragrafo: string, idx: number) => {
    const t = window.getSelection()?.toString() ?? "";
    if (!t.trim()) return;
    const patch = pecaDeTrecho(tipo, t, paragrafo, idx) as Partial<ConfigRevista> | null;
    if (!patch) return;
    if (tipo === "momento" && cfg.momento_etiqueta.trim()) delete patch.momento_etiqueta;
    window.getSelection()?.removeAllRanges();
    setEscolher(null);
    comDesfazer(tipo, patch, "texto", `Usado do parágrafo ${idx + 1}.`);
  };

  const apagar = (tipo: TipoTrecho) => comDesfazer(
    tipo,
    tipo === "pull_quote" ? { pull_quote: "" } : { momento_activo: false, momento_valor: "", momento_descricao: "" },
    null, tipo === "pull_quote" ? "Frase em grande apagada." : "Número da semana apagado.",
  );

  const Onde = ({ tipo }: { tipo: TipoTrecho }) => {
    const campo = tipo === "pull_quote" ? "pull_quote_posicao" : "momento_posicao";
    const pos = posicaoValida(Number(cfg[campo] ?? 99), pars.length);
    return (
      <label className="flex items-center gap-2 text-[12px] text-muted-foreground">
        Onde aparece
        <select
          className="rounded-md border border-input bg-background px-2 py-1 text-[12px] text-foreground" disabled={bloqueado}
          value={pos} onChange={(e) => editar({ [campo]: Number(e.target.value) } as Partial<ConfigRevista>)}
        >
          <option value={0}>Antes do 1.º parágrafo</option>
          {pars.map((_, i) => <option key={i} value={i + 1}>Depois do parágrafo {i + 1}</option>)}
        </select>
      </label>
    );
  };

  const Cabecalho = ({ tipo, titulo, Icone, tem }: { tipo: TipoTrecho; titulo: string; Icone: typeof Quote; tem: boolean }) => (
    <div className="flex flex-wrap items-center gap-2">
      <Icone className="h-4 w-4 text-primary" />
      <span className="text-[13px] font-semibold text-foreground">{titulo}</span>
      {tem && origens[tipo] && (
        <span className="rounded-full border border-border px-2 py-0.5 text-[10.5px] font-semibold text-muted-foreground">{ROTULO_ORIGEM[origens[tipo] as OrigemPeca]}</span>
      )}
      <span className="ml-auto flex gap-1.5">
        <button type="button" disabled={bloqueado || pars.length === 0} onClick={() => setEscolher(escolher === tipo ? null : tipo)}
          className={`inline-flex items-center gap-1 rounded-md border px-2.5 py-1 text-[12px] font-semibold disabled:opacity-40 ${escolher === tipo ? "border-primary bg-primary/10 text-primary" : "border-border text-foreground"}`}>
          <TextSelect className="h-3.5 w-3.5" /> Escolher do texto
        </button>
        <button type="button" disabled={bloqueado || !tem} onClick={() => apagar(tipo)}
          className="inline-flex items-center gap-1 rounded-md border border-border px-2.5 py-1 text-[12px] font-semibold text-muted-foreground hover:text-destructive disabled:opacity-40">
          <Trash2 className="h-3.5 w-3.5" /> Apagar
        </button>
      </span>
    </div>
  );

  const Escolha = ({ tipo }: { tipo: TipoTrecho }) => escolher !== tipo ? null : (
    <div className="space-y-1.5 rounded-lg border border-primary/40 bg-primary/5 p-2.5">
      <p className="text-[12px] font-semibold text-foreground">
        Seleciona {tipo === "momento" ? "o número" : "a frase"} com o rato e carrega em «Usar seleção».
      </p>
      {pars.map((p, i) => (
        <div key={i} className="flex items-start gap-2 rounded-md bg-background p-2">
          <span className="mt-0.5 text-[11px] font-bold text-muted-foreground">{i + 1}</span>
          <p className="flex-1 select-text text-[13px] leading-5 text-foreground">{p}</p>
          <button type="button" onMouseDown={(e) => e.preventDefault()} onClick={() => usarSelecao(tipo, p, i)}
            className="shrink-0 rounded-md bg-primary px-2 py-1 text-[11.5px] font-semibold text-primary-foreground">Usar seleção</button>
        </div>
      ))}
    </div>
  );

  const temQuote = !!cfg.pull_quote.trim();
  const temMomento = cfg.momento_activo && !!cfg.momento_valor.trim();

  return (
    <div className="space-y-3">
      <div className="space-y-2 rounded-xl border border-border bg-background p-3">
        <Cabecalho tipo="pull_quote" titulo="Frase em grande" Icone={Quote} tem={temQuote} />
        <Escolha tipo="pull_quote" />
        <textarea className={input} rows={2} disabled={bloqueado} aria-label="Frase em grande" placeholder="Uma frase da crónica que mereça sair em grande."
          value={cfg.pull_quote} onChange={(e) => aMao("pull_quote", { pull_quote: e.target.value })} />
        {temQuote && <Onde tipo="pull_quote" />}
      </div>

      <div className="space-y-2 rounded-xl border border-border bg-background p-3">
        <Cabecalho tipo="momento" titulo="Número da semana" Icone={Hash} tem={temMomento} />
        <Escolha tipo="momento" />
        <div className="grid gap-2 sm:grid-cols-[1fr_140px]">
          <input className={input} disabled={bloqueado} aria-label="Etiqueta do número" placeholder="O número da semana"
            value={cfg.momento_etiqueta} onChange={(e) => aMao("momento", { momento_etiqueta: e.target.value })} />
          <input className={input} disabled={bloqueado} aria-label="Número" placeholder="19%"
            value={cfg.momento_valor} onChange={(e) => aMao("momento", { momento_valor: e.target.value, momento_activo: !!e.target.value.trim() })} />
        </div>
        <textarea className={input} rows={2} disabled={bloqueado} aria-label="Descrição do número" placeholder="O que este número significa."
          value={cfg.momento_descricao} onChange={(e) => aMao("momento", { momento_descricao: e.target.value })} />
        {temMomento && <Onde tipo="momento" />}
      </div>
    </div>
  );
}
