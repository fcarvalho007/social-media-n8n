import { useState } from "react";
import { useServerFn } from "@tanstack/react-start";
import { Check, ExternalLink, Loader2, Pencil, Sparkles, Save } from "lucide-react";
import { toast } from "sonner";
import { Button } from "@/components/ui/button";
import { Textarea } from "@/components/ui/textarea";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle } from "@/components/ui/alert-dialog";
import type { EstadoBriefEdicao } from "@/lib/brief.functions";
import { gerarBriefFn, guardarConteudoBriefFn, aprovarDestaqueComporFn } from "@/lib/brief.functions";
export interface AccoesBriefs { recarregar: () => void; registar: (accao: string, detalhe?: unknown) => void }
export function BriefsDaEdicao({ briefs, bloqueado, accoes, carregando, edicaoId }: { briefs: EstadoBriefEdicao[]; bloqueado: boolean; carregando?: boolean; accoes: AccoesBriefs; edicaoId: string }) {
 const gerar=useServerFn(gerarBriefFn),guardar=useServerFn(guardarConteudoBriefFn),aprovar=useServerFn(aprovarDestaqueComporFn);
 const [revisao,setRevisao]=useState<EstadoBriefEdicao|null>(null),[em30,setEm30]=useState(""),[implicacoes,setImplicacoes]=useState(""),[leitura,setLeitura]=useState(""),[ocupado,setOcupado]=useState<string|null>(null),[confirmacao,setConfirmacao]=useState<EstadoBriefEdicao|null>(null),[alterado,setAlterado]=useState(false);
 const abrir=(b:EstadoBriefEdicao)=>{setRevisao(b);setEm30(b.em30.join("\n\n"));setImplicacoes(b.porqueInteressa.map(i=>`${i.rotulo} | ${i.texto}`).join("\n\n"));setLeitura(b.leitura);setAlterado(false);};
 const executar=async(b:EstadoBriefEdicao,tipo:"gerar"|"guardar"|"aprovar")=>{
  setOcupado(b.briefId);
  try {
   if(tipo==="gerar") {const r=await gerar({data:{id:b.briefId,componente:"tudo",intencao:"normal"}});if(r.erro)throw new Error(r.erro);toast.success("Brief gerado para revisão.");}
   if(tipo==="guardar") {await guardar({data:{id:b.briefId,em30:em30.split(/\n\s*\n/),porqueInteressa:implicacoes.split(/\n\s*\n/).map(s=>{const [rotulo,...texto]=s.split("|");return{rotulo:rotulo.trim(),texto:texto.join("|").trim()};}),leitura,fonteUrl:b.fonteUrl??"",publisher:b.fontePublisher??""}});setAlterado(false);setRevisao(null);toast.success("Revisão guardada. Aprovação necessária.");}
   if(tipo==="aprovar") {await aprovar({data:{edicaoId,id:b.briefId,texto:leitura}});setRevisao(null);toast.success("Brief aprovado.");}
   accoes.registar(tipo==="gerar"?"Gerou Brief":tipo==="guardar"?"Reviu Brief":"Aprovou Brief",{brief:b.slug});accoes.recarregar();
  }catch(e){toast.error(e instanceof Error?e.message:"Não foi possível concluir.");}finally{setOcupado(null);setConfirmacao(null);}
 };
 const destaques=briefs.filter(b=>b.papel==="destaque").slice(0,3);
 return <div className="space-y-2">
 {carregando&&<Loader2 className="animate-spin" size={16}/>}
 {destaques.map(b=><div key={b.briefId} className="border-b border-border py-3">
 <div className="flex flex-wrap items-start justify-between gap-2"><p className="min-w-0 flex-1 text-sm font-semibold">{b.titulo}</p><span className="text-xs text-muted-foreground">{ocupado===b.briefId?"A trabalhar…":b.estado==="publicado"?"Publicado":b.aprovada?"Aprovado":b.estado==="por_gerar"?"Por gerar":b.estado==="erro"?"Erro":b.estado==="a_gerar"?"Em geração":"Por rever"}</span></div>
 {b.erro&&<p className="mt-1 text-xs text-destructive">{b.erro}</p>}
 <div className="mt-2 flex flex-wrap gap-2">
 <Button variant="outline" size="sm" disabled={bloqueado||!!ocupado||b.estado==="publicado"} onClick={()=>setConfirmacao(b)}><Sparkles size={14}/>Gerar Brief</Button>
 <Button variant="outline" size="sm" disabled={!!ocupado||b.estado==="por_gerar"||b.estado==="a_gerar"} onClick={()=>abrir(b)}><Pencil size={14}/>Rever / aprovar</Button>
 {b.estado==="publicado"&&b.aprovada&&<Button variant="ghost" size="sm" asChild><a href={`/brief/${b.slug}`} target="_blank" rel="noopener noreferrer"><ExternalLink size={14}/>Abrir Brief</a></Button>}
 </div></div>)}
 <Dialog open={!!revisao} onOpenChange={open=>{if(!open)setRevisao(null);}}><DialogContent className="max-h-[85vh] max-w-2xl overflow-y-auto"><DialogHeader><DialogTitle>{revisao?.titulo}</DialogTitle></DialogHeader>
 <label className="space-y-1 text-sm">Em 30 segundos<Textarea value={em30} rows={5} disabled={bloqueado} onChange={e=>{setEm30(e.target.value);setAlterado(true);}}/></label>
 <label className="space-y-1 text-sm">Porque interessa<Textarea value={implicacoes} rows={5} disabled={bloqueado} onChange={e=>{setImplicacoes(e.target.value);setAlterado(true);}}/></label>
 <label className="space-y-1 text-sm">A minha leitura<Textarea value={leitura} rows={6} disabled={bloqueado} onChange={e=>{setLeitura(e.target.value);setAlterado(true);}}/></label>
 {revisao?.fonteUrl&&<a href={revisao.fonteUrl} target="_blank" rel="noopener noreferrer" className="text-sm text-primary">Fonte original ↗</a>}
 <div className="flex flex-wrap gap-2"><Button variant="outline" disabled={bloqueado||!!ocupado||!alterado} onClick={()=>{if(revisao)void executar(revisao,"guardar");}}><Save size={14}/>Guardar revisão</Button><Button disabled={bloqueado||!!ocupado||alterado||!leitura.trim()||!em30.trim()||!implicacoes.trim()||revisao?.aprovada} onClick={()=>{if(revisao)void executar(revisao,"aprovar");}}><Check size={14}/>{revisao?.aprovada?"Aprovado":"Aprovar Brief"}</Button></div>
 </DialogContent></Dialog>
 <AlertDialog open={!!confirmacao} onOpenChange={open=>{if(!open)setConfirmacao(null);}}><AlertDialogContent><AlertDialogHeader><AlertDialogTitle>Gerar este Brief?</AlertDialogTitle><AlertDialogDescription>Esta ação utiliza IA e pode ter custo. A síntese e as implicações serão propostas para revisão; nenhuma nova leitura assinada será publicada automaticamente.</AlertDialogDescription></AlertDialogHeader><AlertDialogFooter><AlertDialogCancel>Cancelar</AlertDialogCancel><AlertDialogAction onClick={e=>{e.preventDefault();if(confirmacao)void executar(confirmacao,"gerar");}}>Confirmar geração</AlertDialogAction></AlertDialogFooter></AlertDialogContent></AlertDialog>
 </div>;
}
