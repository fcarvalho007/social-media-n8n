import { z } from "zod";

export const slideSchema = z
  .object({
    titulo: z.string().trim().min(1).max(100),
    texto: z.string().trim().min(1).max(360),
    fontes: z.array(z.number().int().positive()).max(12),
  })
  .strict();

export const carrosselSchema = z
  .object({
    slides: z.array(slideSchema).min(6).max(8),
    legenda: z.string().trim().min(1).max(2200),
  })
  .strict();

export type Carrossel = z.infer<typeof carrosselSchema>;
export interface FonteCronica {
  edicaoId: string;
  numero: number;
  titulo: string;
  paragrafos: string[];
  url: string;
  origem: "snapshot" | "historico_actual";
  hash: string;
}
export interface RascunhoCarrossel {
  fonte: FonteCronica;
  carrossel: Carrossel;
  versao: number;
  actualizadoEm: string | null;
}

export function validarCarrossel(valor: unknown, fonte: FonteCronica): Carrossel {
  const resultado = carrosselSchema.safeParse(valor);
  if (!resultado.success) {
    throw new Error(
      "A proposta precisa de 6–8 slides, títulos até 100 caracteres e textos até 360. Ajusta o conteúdo e tenta novamente.",
    );
  }
  for (const [i, slide] of resultado.data.slides.entries()) {
    if (i > 0 && i < resultado.data.slides.length - 1 && !slide.fontes.length) {
      throw new Error(`O slide ${i + 1} precisa de pelo menos um parágrafo de referência.`);
    }
    if (slide.fontes.some((n) => n > fonte.paragrafos.length)) {
      throw new Error(`O slide ${i + 1} refere um parágrafo que não existe na crónica.`);
    }
  }
  return resultado.data;
}

export function fonteHistoricaPermitida(fonte: FonteCronica, aceite: boolean): void {
  if (fonte.origem === "historico_actual" && !aceite) {
    throw new Error(
      "Esta edição não guardou a crónica integral no envio. Confirma a utilização do texto atualmente guardado antes de gerar.",
    );
  }
}
